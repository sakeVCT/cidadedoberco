function render(list) { const container = document.getElementById('results'); const hoje = new Date(); hoje.setHours(0,0,0,0); container.innerHTML = list.length === 0 ? "

Sem resultados.

" : list.map(item => { // Só validamos a data se o campo premium estiver ATIVO no admin let isValido = false; if (item.premium === true || item.premium === "true") { isValido = true; // É premium por definição // Mas se tiver data, verificamos se já expirou if (item.data_premium) { const exp = new Date(item.data_premium); exp.setHours(0,0,0,0); if (exp < hoje) isValido = false; } } const fotoHtml = item.image ? `` : ``; return `
${fotoHtml}
${isValido ? ` Seleção Premium` : ''}
${item.categories.join(', ')}

${item.title}

${item.address ? ` ${item.address}` : ''} ${item.email_cliente ? ` ${item.email_cliente}` : ''} ${item.website ? ` Website` : ''}
${(isValido && item.whatsapp) ? `
` : ''}
`; }).join(''); }
